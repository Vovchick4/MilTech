# Урок №161. Типові помилки та великий повний приклад

## Частина 1. Типові помилки в ООП

### Помилка 1: `brand = brand;` (не змінює поле)

```c++
class Car {
    string brand;
public:
    Car(string brand) {
        brand = brand;   // ❌ присвоєння параметру самого себе!
    }
};
```

`brand = brand;` — тут обидва імена означають **параметр** (він «перекриває» поле). Поле лишається неініціалізованим.

Правильно:
```c++
Car(string brand)
    : brand(brand)      // ✅ список ініціалізації
{
}
```
або:
```c++
Car(string brand) {
    this->brand = brand;   // ✅ this->brand — поле
}
```

### Помилка 2: батьківський клас без default constructor, дочірній не викликає його

```c++
class Animal {
public:
    Animal(string name) {}   // немає конструктора без параметрів
};

class Dog : public Animal {
public:
    Dog() {}   // ❌ помилка: не можна викликати Animal() — не існує
};
```

Правильно:
```c++
class Dog : public Animal {
public:
    Dog(string name)
        : Animal(name)   // ✅ явно викликаємо Animal(string)
    {}
};
```

### Помилка 3: забули `virtual`

```c++
class Animal {
public:
    void makeSound() {}   // ❌ без virtual
};

class Dog : public Animal {
public:
    void makeSound() {}
};

Animal* pet = new Dog();
pet->makeSound();   // викличеться Animal::makeSound(), а не Dog::makeSound()
```

Правильно:
```c++
class Animal {
public:
    virtual void makeSound() {}   // ✅ virtual
};
```

### Помилка 4: забули `override`

```c++
class Dog : public Animal {
public:
    void makesound() {}   // ❌ помилка в назві (регістр)
    // компілятор мовчить, але поліморфізм не працює
};
```

Правильно:
```c++
class Dog : public Animal {
public:
    void makeSound() override {}   // ✅ override знайшов би помилку назви
};
```

### Помилка 5: базовий деструктор не `virtual`

```c++
class Animal {
public:
    ~Animal() {}   // ❌ не virtual
};

class Dog : public Animal {
public:
    ~Dog() { /* звільнення ресурсів */ }
};

Animal* pet = new Dog();
delete pet;   // ❌ викликається лише ~Animal() — витікають ресурси Dog
```

Правильно:
```c++
class Animal {
public:
    virtual ~Animal() {}   // ✅
};
```

### Помилка 6: неправильно використаний `protected`

```c++
class Animal {
public:
    string name;   // ❌ public — кожен може змінити
};

// Насправді, якщо треба лише нащадкам:
class Animal {
protected:
    string name;   // ✅ protected
};
```

Але занадто багато `protected` — теж погано. Якщо можна, краще `private` + getter.

### Помилка 7: наслідування там, де краще композиція

```c++
class Engine {
public:
    void start() {}
};

class Car : public Engine {   // ❌ автомобіль НЕ є двигуном
};
```

Правильно:
```c++
class Car {
private:
    Engine engine;   // ✅ автомобіль МАЄ двигун (композиція)
public:
    void start() { engine.start(); }
};
```

## Частина 2. Великий повний приклад

### «Система транспортних засобів»

Цей приклад використовує ВСІ вивчені концепції:

```
                 Vehicle
              abstract class
                    │
        ┌───────────┼───────────┐
        ↓           ↓           ↓
       Car      Motorcycle     Truck
```

- `private` поля;
- конструктори;
- список ініціалізації;
- наслідування;
- `protected`/`private`;
- `virtual`;
- `override`;
- абстрактний клас;
- поліморфізм;
- `virtual` деструктор;
- геттери;
- композиція.

```c++
#include <iostream>
#include <string>
using namespace std;

// ============ Engine (композиція для Car) ============

class Engine {
private:
    string type;
    int horsepower;

public:
    Engine(string t, int hp)
        : type(t), horsepower(hp) {}

    string getType() const { return type; }
    int getHorsepower() const { return horsepower; }

    void start() {
        cout << "  [Engine " << type << " (" << horsepower
             << " hp) запущено]\n";
    }
};

// ============ Абстрактний клас Vehicle ============

class Vehicle {
protected:
    string brand;
    string model;
    int year;

public:
    Vehicle(string b, string m, int y)
        : brand(b), model(m), year(y) {}

    // Чисто віртуальні методи (абстракція)
    virtual void start() = 0;
    virtual void stop() = 0;

    // Реалізований метод (усі класи успадковують)
    void printInfo() const {
        cout << brand << " " << model << " (" << year << ")\n";
    }

    // Геттери
    string getBrand() const { return brand; }
    string getModel() const { return model; }
    int getYear() const { return year; }

    // Віртуальний деструктор — обов'язково!
    virtual ~Vehicle() {
        cout << "  Vehicle (базовий) знищено\n";
    }
};

// ============ Car (наслідування + композиція) ============

class Car : public Vehicle {
private:
    Engine engine;   // композиція: Car має Engine
    int doors;

public:
    Car(string b, string m, int y, string engineType, int hp, int d)
        : Vehicle(b, m, y),        // ініціалізація базового класу
          engine(engineType, hp),  // ініціалізація Engine (композиція)
          doors(d)                 // ініціалізація власного поля
    {}

    void start() override {
        cout << "Автомобіль ";
        printInfo();
        engine.start();   // делегуємо запуску двигуна
    }

    void stop() override {
        cout << "Автомобіль " << model << " зупинено\n";
    }

    int getDoors() const { return doors; }
};

// ============ Motorcycle ============

class Motorcycle : public Vehicle {
private:
    bool hasSidecar;

public:
    Motorcycle(string b, string m, int y, bool sidecar)
        : Vehicle(b, m, y), hasSidecar(sidecar) {}

    void start() override {
        cout << "Мотоцикл ";
        printInfo();
        cout << "  [Мотор запущено]\n";
    }

    void stop() override {
        cout << "Мотоцикл " << model << " зупинено\n";
    }

    bool hasSidecar() const { return hasSidecar; }
};

// ============ Truck ============

class Truck : public Vehicle {
private:
    double cargoCapacity;   // тонни

public:
    Truck(string b, string m, int y, double capacity)
        : Vehicle(b, m, y), cargoCapacity(capacity) {}

    void start() override {
        cout << "Вантажівка ";
        printInfo();
        cout << "  [Дизель запущено]\n";
    }

    void stop() override {
        cout << "Вантажівка " << model << " зупинена\n";
    }

    double getCargoCapacity() const { return cargoCapacity; }
};

// ============ Головна програма: поліморфізм ============

int main() {
    // Масив вказівників на базовий клас — поліморфізм
    Vehicle* vehicles[] = {
        new Car("Toyota", "Camry", 2020, "V4", 200, 4),
        new Car("BMW", "X5", 2021, "V6", 300, 5),
        new Motorcycle("Honda", "CBR", 2019, false),
        new Truck("Volvo", "FH16", 2018, 20.0)
    };

    const int count = 4;

    // Один цикл для всіх типів — потужність поліморфізму
    for (int i = 0; i < count; i++) {
        cout << "--- Запуск транспортного засобу #" << (i + 1) << " ---\n";
        vehicles[i]->start();
        vehicles[i]->stop();
        cout << "\n";
    }

    // Доступ до специфічних методів через RTTI можна, але частіше
    // використовують лише загальний інтерфейс (як вище).

    // Правильне очищення пам'яті
    // (virtual destructor гарантує, що знищаться і нащадки)
    for (int i = 0; i < count; i++) {
        delete vehicles[i];
        cout << "  (об'єкт #" << (i + 1) << " звільнено)\n";
    }
    return 0;
}
```

### Розбір великого прикладу

1. **Абстрактний клас**: `Vehicle` має два чисто віртуальні методи `start()` і `stop()`. Створити `Vehicle vehicle;` неможливо.
2. **Наслідування**: `Car`, `Motorcycle`, `Truck` — нащадки `Vehicle`. Кожен реалізує `start()` і `stop()` через `override`.
3. **Композиція**: `Car` має поле `Engine`. `Engine` — незалежний клас, яким `Car` користується.
4. **Список ініціалізації**: у `Car` вказані базовий клас, `engine`, `doors`.
5. **`private`/`protected`**: поля базового класу (brand, model, year) — `protected`, поля дочірніх — `private`.
6. **Геттери**: `getBrand()`, `getModel()`, `getEngineType()` тощо.
7. **Поліморфізм**: масив `Vehicle*` містить різні типи. Цикл викликає правильні `start()`/`stop()` залежно від реального типу.
8. **`virtual` деструктор**: у кінці `delete` правильно знищує кожен нащадок.
